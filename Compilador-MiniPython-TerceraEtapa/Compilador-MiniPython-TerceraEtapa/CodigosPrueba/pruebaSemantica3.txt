# Código de prueba adicional con funciones y variables

def add(a, b, c):
    return a + b + c

def square(num):
    return num * num

z = 20  # Variable global

# Definición de una función que usa variables locales y llamadas a otras funciones
def calculate(x, y):
    total = add(x, y, z)
    squared_total = square(total)
    return squared_total

result = calculate(5, 15)  # Llamada a función con dos parámetros
squared_z = square(z)  # Llamada a función con un solo parámetro

# Llamadas correctas a las funciones con la cantidad adecuada de parámetros
print("Resultado:", result)
print("Cuadrado de z:", squared_z)
